## 🎯 Project Overview

**Objective**: Binary classification of molecular activity using Message Passing Neural Networks (MPNNs) enhanced with RDKit molecular descriptors

**Approach**: Ensemble learning with 10 models trained with different random seeds, incorporating both graph-based features and traditional molecular descriptors

**Key Results**:

- **With RDKit Features**: AUPRC: 0.665, AUROC: 0.731
- **Baseline (Graph Only)**: AUPRC: 0.581, AUROC: 0.654
- **Improvement**: +14.5% AUPRC, +11.8% AUROC
- Ensemble of 10 models
- 25 test samples evaluated

## 🧬 RDKit Features Integration

### 1. Available RDKit Feature Generators

Chemprop supports multiple feature generators:

| Feature Generator     | Description                     | Size          | Normalization  |
| --------------------- | ------------------------------- | ------------- | -------------- |
| `morgan`              | Morgan fingerprints (binary)    | 2048 bits     | Not required   |
| `morgan_count`        | Morgan fingerprints (counts)    | 2048 bits     | Recommended    |
| `rdkit_2d`            | RDKit 2D descriptors            | ~200 features | Recommended    |
| `rdkit_2d_normalized` | RDKit 2D normalized descriptors | ~200 features | Pre-normalized |

### 2. RDKit 2D Normalized Features (Recommended)

**Key advantages**:

- Pre-normalized molecular descriptors
- Includes: LogP, molecular weight, TPSA, rotatable bonds, aromatic rings, etc.
- Stable numerical values
- No additional scaling required

**Usage in training**:

```bash
--features_generator rdkit_2d_normalized --no_features_scaling
```

### 3. Alternative RDKit Options

**Standard RDKit 2D features**:

```bash
--features_generator rdkit_2d --features_scaling
```

**Morgan fingerprints**:

```bash
--features_generator morgan --no_features_scaling
```

## 📈 Model Evaluation

### 1. Create Enhanced Evaluation Script

**Create `performance_evaluation_enhanced.py`**:

```python
import csv
import pandas as pd
import numpy as np
from sklearn.metrics import precision_recall_curve, auc, roc_curve, roc_auc_score, classification_report
import matplotlib.pyplot as plt
import seaborn as sns

def calculate_metrics_enhanced(pred_file, true_file, model_name="Model"):
    """Calculate comprehensive metrics for model evaluation"""

    # Load predictions and true labels
    pred_df = pd.read_csv(pred_file)
    true_df = pd.read_csv(true_file)

    # Extract scores and labels (ensure numeric)
    y_scores = pred_df['ACTIVITY'].astype(float).values
    y_true = true_df['ACTIVITY'].astype(float).values

    # Calculate AUPRC
    precision, recall, _ = precision_recall_curve(y_true, y_scores)
    auprc = auc(recall, precision)

    # Calculate AUROC
    fpr, tpr, _ = roc_curve(y_true, y_scores)
    auroc = roc_auc_score(y_true, y_scores)

    # Binary predictions (using 0.5 threshold)
    y_pred_binary = (y_scores > 0.5).astype(int)

    # Additional statistics
    num_samples = len(y_true)
    class_dist = {int(label): int(count) for label, count in zip(*np.unique(y_true, return_counts=True))}
    mean_pred = np.mean(y_scores)
    std_pred = np.std(y_scores)

    # Classification metrics
    from sklearn.metrics import accuracy_score, f1_score, precision_score, recall_score
    accuracy = accuracy_score(y_true, y_pred_binary)
    f1 = f1_score(y_true, y_pred_binary)
    precision_score_val = precision_score(y_true, y_pred_binary)
    recall_score_val = recall_score(y_true, y_pred_binary)

    # Print results
    print(f"\n{model_name} Performance Results")
    print("=" * (len(model_name) + 20))
    print(f"AUPRC: {auprc:.4f}")
    print(f"AUROC: {auroc:.4f}")
    print(f"Accuracy: {accuracy:.4f}")
    print(f"F1 Score: {f1:.4f}")
    print(f"Precision: {precision_score_val:.4f}")
    print(f"Recall: {recall_score_val:.4f}")
    print(f"Num_Samples: {num_samples}")
    print(f"Class_Distribution: {class_dist}")
    print(f"Mean_Prediction: {mean_pred:.4f}")
    print(f"Std_Prediction: {std_pred:.4f}")

    # Save results to file
    results_file = f'performance_results_{model_name.lower().replace(" ", "_")}.txt'
    with open(results_file, 'w') as f:
        f.write(f"{model_name} Performance Results\n")
        f.write("=" * (len(model_name) + 20) + "\n\n")
        f.write(f"AUPRC: {auprc}\n")
        f.write(f"AUROC: {auroc}\n")
        f.write(f"Accuracy: {accuracy}\n")
        f.write(f"F1 Score: {f1}\n")
        f.write(f"Precision: {precision_score_val}\n")
        f.write(f"Recall: {recall_score_val}\n")
        f.write(f"Num_Samples: {num_samples}\n")
        f.write(f"Class_Distribution: {class_dist}\n")
        f.write(f"Mean_Prediction: {mean_pred}\n")
        f.write(f"Std_Prediction: {std_pred}\n")

    return {
        'auprc': auprc,
        'auroc': auroc,
        'accuracy': accuracy,
        'f1': f1,
        'precision': precision_score_val,
        'recall': recall_score_val,
        'precision_curve': precision,
        'recall_curve': recall,
        'fpr': fpr,
        'tpr': tpr,
        'y_scores': y_scores,
        'y_true': y_true
    }

def plot_comparison(results_dict, save_path='model_comparison.png'):
    """Plot comparison between different models"""

    fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 12))

    # PR Curves
    for model_name, results in results_dict.items():
        ax1.plot(results['recall_curve'], results['precision_curve'],
                label=f'{model_name} (AUPRC={results["auprc"]:.3f})', linewidth=2)
    ax1.set_xlabel('Recall')
    ax1.set_ylabel('Precision')
    ax1.set_title('Precision-Recall Curves')
    ax1.legend()
    ax1.grid(True, alpha=0.3)

    # ROC Curves
    for model_name, results in results_dict.items():
        ax2.plot(results['fpr'], results['tpr'],
                label=f'{model_name} (AUROC={results["auroc"]:.3f})', linewidth=2)
    ax2.plot([0, 1], [0, 1], 'k--', alpha=0.5, label='Random')
    ax2.set_xlabel('False Positive Rate')
    ax2.set_ylabel('True Positive Rate')
    ax2.set_title('ROC Curves')
    ax2.legend()
    ax2.grid(True, alpha=0.3)

    # Metrics Comparison
    metrics = ['auprc', 'auroc', 'accuracy', 'f1']
    x = np.arange(len(metrics))
    width = 0.35

    models = list(results_dict.keys())
    for i, model in enumerate(models):
        values = [results_dict[model][metric] for metric in metrics]
        ax3.bar(x + i*width, values, width, label=model, alpha=0.8)

    ax3.set_xlabel('Metrics')
    ax3.set_ylabel('Score')
    ax3.set_title('Performance Metrics Comparison')
    ax3.set_xticks(x + width/2)
    ax3.set_xticklabels(metrics)
    ax3.legend()
    ax3.grid(True, alpha=0.3)

    # Prediction Distribution
    for model_name, results in results_dict.items():
        ax4.hist(results['y_scores'], alpha=0.6, label=f'{model_name}', bins=20)
    ax4.set_xlabel('Prediction Score')
    ax4.set_ylabel('Frequency')
    ax4.set_title('Prediction Score Distribution')
    ax4.legend()
    ax4.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig(save_path, dpi=300, bbox_inches='tight')
    plt.show()

# Run evaluation
if __name__ == "__main__":
    # Evaluate RDKit-enhanced model
    print("Evaluating RDKit-enhanced ensemble...")
    rdkit_results = calculate_metrics_enhanced("test_20_results_rdkit.csv", "test_20.csv", "RDKit Enhanced")

    # Evaluate baseline model (if available)
    try:
        print("\nEvaluating baseline ensemble...")
        baseline_results = calculate_metrics_enhanced("test_20_results_baseline.csv", "test_20.csv", "Baseline")

        # Compare models
        results_comparison = {
            'RDKit Enhanced': rdkit_results,
            'Baseline': baseline_results
        }
        plot_comparison(results_comparison)

    except FileNotFoundError:
        print("Baseline results not found. Plotting RDKit results only...")
        plot_comparison({'RDKit Enhanced': rdkit_results})
```

from sklearn.metrics import precision_recall_curve, auc, roc_curve, roc_auc_score
import matplotlib.pyplot as plt

print("✅ Libraries imported successfully!")

# Cell 2: Load Data

try:

    print("✅ RDKit predictions loaded")

# Load true labels
true_labels = pd.read_csv('test_20.csv')

# Extract data
y_scores = rdkit_pred['ACTIVITY'].values
y_true = true_labels['ACTIVITY'].values

print(f"Data shape: {len(y_scores)} samples")
print(f"Class distribution: {np.bincount(y_true)}")

# Cell 3: Calculate Metrics
precision, recall, _ = precision_recall_curve(y_true, y_scores)
auprc = auc(recall, precision)
fpr, tpr, _ = roc_curve(y_true, y_scores)

print(f"AUROC: {auroc:.4f}")

fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 10))

ax1.plot(recall, precision, 'b-', linewidth=2, label=f'AUPRC = {auprc:.3f}')

ax1.set_xlabel('Recall')

ax2.fill_between(fpr, tpr, alpha=0.3)
ax2.set_xlabel('False Positive Rate')
ax2.set_ylabel('True Positive Rate')
ax2.set_title('ROC Curve')
ax2.legend()
ax2.grid(True, alpha=0.3)

# Prediction Distribution
ax3.hist(y_scores, bins=20, alpha=0.7, color='skyblue', edgecolor='black')

ax3.set_xlabel('Prediction Score')

ax3.set_title('Prediction Score Distribution')
ax3.legend()
ax3.grid(True, alpha=0.3)

# Threshold analysis
thresholds = np.arange(0.1, 0.9, 0.1)
for threshold in thresholds:
    y_pred = (y_scores > threshold).astype(int)
    accuracy = np.mean(y_pred == y_true)
    tp = np.sum((y_pred == 1) & (y_true == 1))
    fp = np.sum((y_pred == 1) & (y_true == 0))
    tn = np.sum((y_pred == 0) & (y_true == 0))
    fn = np.sum((y_pred == 0) & (y_true == 1))

    precision_t = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall_t = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1_t = 2 * (precision_t * recall_t) / (precision_t + recall_t) if (precision_t + recall_t) > 0 else 0

    print(f"Threshold: {threshold:.1f} | Acc: {accuracy:.3f} | Prec: {precision_t:.3f} | Rec: {recall_t:.3f} | F1: {f1_t:.3f}")
